-- =============================================
-- TABELA: usuarios
-- =============================================
CREATE TABLE usuarios (
    id VARCHAR(36) PRIMARY KEY DEFAULT (UUID()),
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    senha VARCHAR(255) NOT NULL,
    metadados JSON DEFAULT ('{}'),
    data_criado TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- =============================================
-- TABELA: projetos
-- =============================================
CREATE TABLE projetos (
    id VARCHAR(36) PRIMARY KEY DEFAULT (UUID()),
    nome VARCHAR(100) NOT NULL,
    descricao TEXT,
    data_criado TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ultima_alteracao TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    id_usuario VARCHAR(36) NOT NULL,
    CONSTRAINT fk_projeto_usuario 
        FOREIGN KEY (id_usuario) 
        REFERENCES usuarios(id) 
        ON DELETE CASCADE
);

-- =============================================
-- TABELA: elementos
-- =============================================
CREATE TABLE elementos (
    id VARCHAR(36) PRIMARY KEY DEFAULT (UUID()),
    nome VARCHAR(100) NOT NULL DEFAULT 'Novo Elemento',
    descricao TEXT,
    conteudo TEXT,
    tipo VARCHAR(20) NOT NULL DEFAULT 'TEXTO',
    coordenadas JSON DEFAULT ('{"x": 0, "y": 0, "width": 100, "height": 100}'),
    relacoes JSON DEFAULT ('[]'),
    data_criado TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    ultima_alteracao TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    id_projeto VARCHAR(36) NOT NULL,
    CONSTRAINT fk_elemento_projeto 
        FOREIGN KEY (id_projeto) 
        REFERENCES projetos(id) 
        ON DELETE CASCADE,
    CONSTRAINT chk_tipo_valido 
        CHECK (tipo IN ('CIRCULO', 'RETANGULO', 'TRIANGULO', 'TEXTO'))
);

-- =============================================
-- ÍNDICES PARA PERFORMANCE
-- =============================================
CREATE INDEX idx_usuarios_email ON usuarios(email);
CREATE INDEX idx_projetos_usuario ON projetos(id_usuario);
CREATE INDEX idx_elementos_projeto ON elementos(id_projeto);
CREATE INDEX idx_elementos_tipo ON elementos(tipo);
